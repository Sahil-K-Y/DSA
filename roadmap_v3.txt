```cpp
      while (!q.empty()) {
          int sz = q.size();
          vector<int> level;
          for (int i = 0; i < sz; i++) {
              TreeNode* node = q.front(); q.pop();
              level.push_back(node->val);
              if (node->left) q.push(node->left);
              if (node->right) q.push(node->right);
          }
          ans.push_back(level);
      }
```

```cpp
  int lengthOfLongestSubstring(string s) {
      unordered_map<char, int> freq;
      int left = 0, maxLen = 0;
      for (int right = 0; right < s.length(); right++) {
          freq[s[right]]++;
          while (freq[s[right]] > 1) {
              freq[s[left]]--;
              left++;
          }
          maxLen = max(maxLen, right - left + 1);
      }
      return maxLen;
  }
```

```cpp
      unordered_map<int, int> mp;
      mp[0] = 1; // Base case for prefix sum
      int sum = 0, count = 0;
      for (int x : nums) {
          sum += x;
          if (mp.find(sum - k) != mp.end()) count += mp[sum - k];
          mp[sum]++;
      }
      return count;
  }
```

  • Each element pushed and popped at most once → Amortized O(N) time complexity.
  • Stores indices (not values) to calculate distances `i - st.top()`.
  - **C++ Scratch Implementation & Syntax Recall:**
```cpp

      int n = temp.size();
      vector<int> ans(n, 0);
      stack<int> st; // Stores indices
      for (int i = 0; i < n; i++) {
          while (!st.empty() && temp[i] > temp[st.top()]) {
              int prevIdx = st.top(); st.pop();
              ans[prevIdx] = i - prevIdx;
          }
          st.push(i);
      }
      return ans;
  }
```

```cpp
          fast = fast->next->next;
          if (slow == fast) return true;
      }
      return false;
  }
```

  • If `nums[low] <= nums[mid]`, left half is sorted. Check if target lies within `[nums[low], nums[mid]]`.

  • Duplicates handling requires shrinking `low++` and `high--` when `nums[low] == nums[mid] == nums[high]`.
  - **C++ Scratch Implementation & Syntax Recall:**
```cpp
  int searchRotated(vector<int>& nums, int target) {
      int low = 0, high = nums.size() - 1;
      while (low <= high) {
          int mid = low + (high - low) / 2;
          if (nums[mid] == target) return mid;
          if (nums[low] <= nums[mid]) { // Left sorted
              if (target >= nums[low] && target < nums[mid]) high = mid - 1;
              else low = mid + 1;
          } else { // Right sorted
              if (target > nums[mid] && target <= nums[high]) low = mid + 1;
              else high = mid - 1;
          }
      }
      return -1;
  }
```

- **Phase:** Dynamic Programming & Sequence DP (Days 086–135)

- **Track B (Daily Spaced Foundation Revision & Theory Recall):**
  - **Category & Focus:** Binary Search — Binary Search in Rotated Sorted Arrays
  - **Theory & Concept Recall:**
  • Array is sorted then rotated at pivot. Key insight: At least one half (`[low..mid]` or `[mid..high]`) is always strictly sorted.
  • If `nums[low] <= nums[mid]`, left half is sorted. Check if target lies within `[nums[low], nums[mid]]`.
  • Otherwise, right half is sorted. Check if target lies within `[nums[mid], nums[high]]`.
  • Decide which half to discard accordingly, maintaining O(log N) search time.
  • Duplicates handling requires shrinking `low++` and `high--` when `nums[low] == nums[mid] == nums[high]`.
  - **C++ Scratch Implementation & Syntax Recall:**
```cpp
  int searchRotated(vector<int>& nums, int target) {
      int low = 0, high = nums.size() - 1;
      while (low <= high) {
          int mid = low + (high - low) / 2;
          if (nums[mid] == target) return mid;
          if (nums[low] <= nums[mid]) { // Left sorted
              if (target >= nums[low] && target < nums[mid]) high = mid - 1;
              else low = mid + 1;
          } else { // Right sorted
              if (target > nums[mid] && target <= nums[high]) low = mid + 1;
              else high = mid - 1;
          }
      }
      return -1;
  }
```

    - [ ] LC 33 Search in Rotated Sorted Array

  - **C++ Scratch Implementation & Syntax Recall:**

```cpp
  int searchRotated(vector<int>& nums, int target) {
      int low = 0, high = nums.size() - 1;
      while (low <= high) {
          int mid = low + (high - low) / 2;
          if (nums[mid] == target) return mid;
          if (nums[low] <= nums[mid]) { // Left sorted
              if (target >= nums[low] && target < nums[mid]) high = mid - 1;
              else low = mid + 1;
          } else { // Right sorted
              if (target > nums[mid] && target <= nums[high]) low = mid + 1;
              else high = mid - 1;
          }
      }
      return -1;
  }
```

```cpp
      stack<int> st; // Stores indices
      for (int i = 0; i < n; i++) {
          while (!st.empty() && temp[i] > temp[st.top()]) {
              int prevIdx = st.top(); st.pop();
              ans[prevIdx] = i - prevIdx;
          }
          st.push(i);
      }
      return ans;
  }
```

```cpp
              left++;
          }
```

```cpp
          vector<int> level;

              level.push_back(node->val);
              if (node->left) q.push(node->left);
```

```cpp
              ans[prevIdx] = i - prevIdx;

          st.push(i);
```

### 📆 Day 157: Lowest Common Ancestor — Binary Tree DFS
- **Phase:** Advanced Trees, Construction, Serialization & LCA (Days 151–165)

- **Track B (Daily Spaced Foundation Revision & Theory Recall):**
  - **Category & Focus:** Binary Search — Binary Search in Rotated Sorted Arrays
  - **Theory & Concept Recall:**
  • Array is sorted then rotated at pivot. Key insight: At least one half (`[low..mid]` or `[mid..high]`) is always strictly sorted.
  • If `nums[low] <= nums[mid]`, left half is sorted. Check if target lies within `[nums[low], nums[mid]]`.
  • Otherwise, right half is sorted. Check if target lies within `[nums[mid], nums[high]]`.
  • Decide which half to discard accordingly, maintaining O(log N) search time.
  • Duplicates handling requires shrinking `low++` and `high--` when `nums[low] == nums[mid] == nums[high]`.
  - **C++ Scratch Implementation & Syntax Recall:**
```cpp
  int searchRotated(vector<int>& nums, int target) {
      int low = 0, high = nums.size() - 1;
      while (low <= high) {
          int mid = low + (high - low) / 2;
          if (nums[mid] == target) return mid;
          if (nums[low] <= nums[mid]) { // Left sorted
              if (target >= nums[low] && target < nums[mid]) high = mid - 1;
              else low = mid + 1;
          } else { // Right sorted
              if (target > nums[mid] && target <= nums[high]) low = mid + 1;
              else high = mid - 1;
          }
      }
      return -1;
  }
```

  • Backtracking Pattern: 3 steps -> (1) Choose candidate, (2) Explore recursively, (3) Un-choose candidate (backtrack).
  • Combinational Search: Use `startIdx` in recursive function to avoid generating duplicate permutations.
  • Time Complexity: Often O(2^N) or O(N!) depending on decision tree branching factor.
  - **C++ Scratch Implementation & Syntax Recall:**

```cpp
  void backtrack(int start, vector<int>& nums, vector<int>& path, vector<vector<int>>& ans) {
      ans.push_back(path);
      for (int i = start; i < nums.size(); i++) {
          path.push_back(nums[i]);
          backtrack(i + 1, nums, path, ans);
          path.pop_back(); // Backtrack step
      }
  }
  vector<vector<int>> subsets(vector<int>& nums) {
      vector<vector<int>> ans; vector<int> path;
      backtrack(0, nums, path, ans);
      return ans;
  }
```

```cpp
          vector<int> level;
          for (int i = 0; i < sz; i++) {
              TreeNode* node = q.front(); q.pop();
              level.push_back(node->val);
              if (node->left) q.push(node->left);
              if (node->right) q.push(node->right);
          }
          ans.push_back(level);
      }
      return ans;
  }
```

```cpp
          while (!st.empty() && temp[i] > temp[st.top()]) {
              int prevIdx = st.top(); st.pop();
              ans[prevIdx] = i - prevIdx;
```
